-- ============================================================ -- crm_0002_leads_updated_at.sql -- Run on the CRM's Supabase project. -- -- Fixes a real, widespread bug just discovered via the sponsor-link -- SQL error: the leads table has no updated_at column at all, only -- created_at. At least 16 separate places in the CRM's own code -- depend on updated_at — including features that predate tonight -- entirely, like stale-lead alerts and lead-leakage detection. None -- of them ever threw an error, because the code always checks -- `if (l.updated_at)` first — they simply silently never triggered -- instead, since that check always failed. This has likely meant: -- - Stale-lead notifications never fired -- - Lead leakage detection always showed zero -- - Lead scoring's recency bonus/penalty never applied -- - Weekly Wins never showed any conversions -- - Data Health's "untouched 30+ days" check never flagged anything -- -- This migration adds the column properly, backfills existing rows, -- and — critically — adds a trigger so it updates itself -- automatically on every future change, rather than relying on the -- application code to remember to set it (which is exactly how this -- gap likely happened in the first place). -- ============================================================ alter table public.leads add column if not exists updated_at timestamptz; -- Backfill: for existing rows, created_at is the best available -- starting point — better than leaving it null and having every -- existing lead look "just updated" or "never updated" incorrectly. update public.leads set updated_at = created_at where updated_at is null; alter table public.leads alter column updated_at set default now(); -- The actual fix that prevents this gap from silently recurring: -- automatically stamps updated_at on every UPDATE, so no application -- code path can forget to set it. create or replace function public.set_leads_updated_at() returns trigger language plpgsql as $$ begin new.updated_at = now(); return new; end; $$; drop trigger if exists trg_leads_updated_at on public.leads; create trigger trg_leads_updated_at before update on public.leads for each row execute function public.set_leads_updated_at();

// ============================================================ // generate-mock-test // Deploy to the STUDENT PORTAL Supabase project (jfrtleleblsphyztyhaf), // NOT the CRM project. // // Two-pass design, matching the original spec: // 1. GENERATE — Claude writes a fresh set of questions from the // human-reviewed template (topic pool, format, difficulty, word // limit) — never the same test twice. // 2. VALIDATE — a second, separate Claude call reviews that output // for correctness, ambiguity, and fit before it's ever shown to // a student. If it finds a real problem, it corrects it directly // rather than just flagging it (student never sees a bad question // OR a regeneration delay). // // Requires the ANTHROPIC_API_KEY secret to be set on THIS project // (Edge Functions → Manage secrets). It is separate from the CRM // project's key even if you reuse the same actual Anthropic account. // ============================================================ import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'; const ANTHROPIC_API_KEY = Deno.env.get('ANTHROPIC_API_KEY'); const SUPABASE_URL = Deno.env.get('SUPABASE_URL'); const SUPABASE_ANON_KEY = Deno.env.get('SUPABASE_ANON_KEY'); const corsHeaders = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type', }; async function callClaude(systemPrompt: string, userPrompt: string): Promise { const res = await fetch('https://api.anthropic.com/v1/messages', { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-api-key': ANTHROPIC_API_KEY!, 'anthropic-version': '2023-06-01', }, body: JSON.stringify({ model: 'claude-sonnet-4-6', max_tokens: 3000, system: systemPrompt, messages: [{ role: 'user', content: userPrompt }], }), }); if (!res.ok) { const errText = await res.text(); throw new Error(`Claude API error (${res.status}): ${errText.slice(0, 300)}`); } const data = await res.json(); const textBlock = (data.content || []).find((b: any) => b.type === 'text'); if (!textBlock) throw new Error('Claude returned no text content'); return textBlock.text; } // Claude sometimes wraps JSON in ```json fences despite instructions — // strip those defensively before parsing, same pattern used elsewhere // in this app's AI features. function extractJson(raw: string): any { const cleaned = raw.replace(/```json\s*|```\s*/g, '').trim(); return JSON.parse(cleaned); } const GENERATION_SCHEMA_NOTE = ` Return ONLY a JSON object, no preamble, no markdown fences, matching exactly: { "passages": [ { "id": "p1", "text": "the full shared passage or audio-transcript text, shown to the student ONCE above its questions" } ], "questions": [ { "prompt": "the individual question text ONLY — never re-paste or repeat the passage text here if it already belongs to a passage above", "passageId": "p1 — ONLY include this field if the question belongs to a shared passage above; omit it entirely for standalone questions", "gradingType": "exact" | "sample", "answer": "the single correct answer — ONLY when gradingType is 'exact'", "sampleAnswer": "a strong model response the student compares their own answer against — ONLY when gradingType is 'sample'", "rubric": ["2-4 short bullet points of what a good answer needs to include — ONLY when gradingType is 'sample'"] } ] } WHEN TO USE "passages": Reading and Listening question types are very often several questions about ONE shared passage or audio transcript (e.g. "questions 1-5 are about the following passage"). In that case: write the shared text ONCE inside "passages", give it an id, and have each of its questions reference that same id via "passageId" — the question's own "prompt" should then be ONLY the actual question itself (e.g. "What is the main topic of paragraph 2?"), NOT the passage restated or re-embedded. If a test has multiple genuinely separate passages (uncommon but possible), add multiple entries to "passages", each with their own questions referencing the matching id. WHEN "passages" IS NOT NEEDED: Speaking prompts, Writing prompts, sentence-building, and any other standalone question with no shared source text should simply omit "passageId" entirely — leave "passages" as an empty array in that case. Use "exact" for questions with one objectively correct answer (multiple choice, true/false/not given, fill-in-the-blank, note/form completion, matching, sentence-level tasks). Use "sample" for open-ended tasks (essays, outlines, spoken responses, summaries, cue-card style answers) where there's no single correct string — give a strong model answer plus a short rubric instead. CRITICAL — plain text only, no markdown formatting whatsoever, in every string field (prompt, passage text, answer, sampleAnswer, rubric entries): no asterisks for bold or italics, no # headers, no markdown lists, no backticks. The student-facing app renders these as plain text, so any markdown syntax shows up as literal, ugly asterisk/hash characters instead of actual formatting. If a passage needs a title or section labels, write them as plain sentences (e.g. "The passage below is titled..." or "Paragraph 1 discusses...") instead of using ** or # to mark them. `.trim(); async function generateQuestions(template: any, testName: string): Promise { const system = `You are writing exam-prep practice questions for a real study-abroad test-prep platform (Step4Progress). You are an expert item-writer for ${testName}, closely following an official, human-approved template. Never copy real exam questions — everything you write must be original. Match the exact question count, format, and difficulty specified. Pick topics from the given pool, varying which ones you use so repeated attempts feel genuinely different — do not always pick the first topics listed.`; const user = `Generate a fresh practice test instance from this template: Title: ${template.title} Question count: ${template.questionCount} Time limit: ${template.timeLimitMinutes} minutes Question type(s): ${(template.questionTypes || []).join(', ')} Topic pool (vary your picks across these): ${(template.topicPool || []).join(' | ')} Difficulty/format note: ${template.difficultyNote || 'N/A'} Word limit / answer format rule: ${template.wordLimitRule || 'N/A'} ${GENERATION_SCHEMA_NOTE}`; const raw = await callClaude(system, user); return extractJson(raw); } async function validateQuestions(generated: any, template: any, testName: string): Promise { const system = `You are a strict quality-control reviewer for ${testName} exam-prep content. You check AI-generated practice questions against a template BEFORE a real student ever sees them. You look for: factually wrong or ambiguous answer keys, questions that don't match the stated difficulty or format, sample answers that are weak or don't actually satisfy their own rubric, and any content that doesn't fit the topic pool or word-limit rule. If everything is genuinely fine, return it unchanged. If you find real problems, FIX them directly in your output — do not just describe the problem.`; const user = `Template requirements: Question count: ${template.questionCount} Question type(s): ${(template.questionTypes || []).join(', ')} Difficulty/format note: ${template.difficultyNote || 'N/A'} Word limit / answer format rule: ${template.wordLimitRule || 'N/A'} Generated questions to review and, if needed, correct: ${JSON.stringify(generated, null, 2)} ${GENERATION_SCHEMA_NOTE} Return the final, corrected (or unchanged) JSON in the exact same schema.`; const raw = await callClaude(system, user); return extractJson(raw); } function isWellFormed(data: any, expectedCount: number): boolean { if (!data || !Array.isArray(data.questions)) return false; if (data.questions.length !== expectedCount) return false; // passages is optional (many question types — Speaking, Writing — // never need a shared passage at all), but if present, each entry // needs a real id and real text. if (data.passages !== undefined) { if (!Array.isArray(data.passages)) return false; if (!data.passages.every((p: any) => typeof p.id === 'string' && p.id.length > 0 && typeof p.text === 'string' && p.text.length > 0)) return false; } return data.questions.every((q: any) => typeof q.prompt === 'string' && q.prompt.length > 0 && (q.gradingType === 'exact' || q.gradingType === 'sample') && (q.gradingType === 'exact' ? typeof q.answer === 'string' && q.answer.length > 0 : typeof q.sampleAnswer === 'string' && q.sampleAnswer.length > 0) ); } Deno.serve(async (req: Request) => { if (req.method === 'OPTIONS') return new Response('ok', { headers: corsHeaders }); try { if (!ANTHROPIC_API_KEY) { return new Response(JSON.stringify({ error: 'ANTHROPIC_API_KEY is not set on this project. Add it under Edge Functions → Manage secrets.' }), { status: 500, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }); } // Require a genuinely logged-in student — this costs real money per // call, so it must never be reachable by an unauthenticated request. const authHeader = req.headers.get('Authorization'); if (!authHeader) { return new Response(JSON.stringify({ error: 'Sign in required.' }), { status: 401, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }); } const supabase = createClient(SUPABASE_URL!, SUPABASE_ANON_KEY!, { global: { headers: { Authorization: authHeader } }, }); const { data: userData, error: userErr } = await supabase.auth.getUser(); if (userErr || !userData?.user) { return new Response(JSON.stringify({ error: 'Sign in required.' }), { status: 401, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }); } const { template, testName } = await req.json(); if (!template || !template.questionCount) { return new Response(JSON.stringify({ error: 'Missing or invalid template.' }), { status: 400, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }); } // Pass 1 — generate. let result = await generateQuestions(template, testName || 'this test'); // Pass 2 — validate/correct. If either pass produced something // malformed, retry generation once before giving up — better to // eat one extra AI call than hand back a broken test. if (isWellFormed(result, template.questionCount)) { const validated = await validateQuestions(result, template, testName || 'this test'); if (isWellFormed(validated, template.questionCount)) { result = validated; } // If validation came back malformed, we keep the pass-1 result // rather than fail outright — it already passed its own // well-formedness check. } else { result = await generateQuestions(template, testName || 'this test'); if (!isWellFormed(result, template.questionCount)) { return new Response(JSON.stringify({ error: "The generator produced an invalid test twice in a row. Please try again — this is usually temporary." }), { status: 502, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }); } } return new Response(JSON.stringify({ questions: result.questions, passages: result.passages || [] }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, }); } catch (err) { return new Response(JSON.stringify({ error: String(err?.message || err) }), { status: 500, headers: { ...corsHeaders, 'Content-Type': 'application/json' }, }); } });
